var ArrayList = Java.type('java.util.ArrayList');
 
var access_token = keycloakSession.tokens().encode(token);
var query = '{me{' +
    'fieldActivities{id}' +
    '}}';

var result = httpGet("GRAPHQL_INTERNAL_BASE_URL" + "?query=" + encodeURIComponent(query)).data;

var user_field_activities = new ArrayList();
JSON.parse(result).data.me.fieldActivities.map(function(org){
    user_field_activities.add(org);
});
token.setOtherClaims("membership", {
    field_activities: user_field_activities
})



/*************
UTILITY FUNCTIONS
*************/

function httpGet(theUrl){
    var con = new java.net.URL(theUrl).openConnection();
    con.requestMethod = "GET";
    con.setRequestProperty("Authorization", "Bearer " + access_token);

    return asResponse(con);
}

function httpPost(theUrl, data, contentType){
    contentType = contentType || "application/json";
    var con = new java.net.URL(theUrl).openConnection();

    con.requestMethod = "POST";
    con.setRequestProperty("Content-Type", contentType);

    // Send post request
    con.doOutput=true;
    write(con.outputStream, data);

    return asResponse(con);
}

function asResponse(con){
    var d = read(con.inputStream);

    return {data : d, statusCode : con.responseCode};
}

function write(outputStream, data){
    var wr = new java.io.DataOutputStream(outputStream);
    wr.writeBytes(data);
    wr.flush();
    wr.close();
}

function read(inputStream){
    var inReader = new java.io.BufferedReader(new java.io.InputStreamReader(inputStream));
    var inputLine;
    var response = new java.lang.StringBuffer();

    while ((inputLine = inReader.readLine()) !== null) {
           response.append(inputLine);
    }
    inReader.close();
    return response.toString();
}
